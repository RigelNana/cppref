{/* source:cpp-ranges-random-access-range:0001 */}
The `random_access_range` concept is a refinement of [`range`](/docs/cpp/library/ranges/range) for which `ranges::begin` returns a model of [`random_access_iterator`](/docs/cpp/library/iterator/random_access_iterator).

### Example

{/* source:cpp-ranges-random-access-range:0002 */}
```cpp
#include <array>
#include <deque>
#include <list>
#include <ranges>
#include <set>
#include <valarray>
#include <vector>
 
template<typename T> concept RAR = std::ranges::random_access_range<T>;
 
int main()
{
    int a[4];
    static_assert(
            RAR<std::vector<int>> and
            RAR<std::vector<bool>> and
            RAR<std::deque<int>> and
            RAR<std::valarray<int>> and
            RAR<decltype(a)> and
        not RAR<std::list<int>> and
        not RAR<std::set<int>> and
            RAR<std::array<std::list<int>,42>>
    );
}
```
