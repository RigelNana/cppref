### Example

{/* source:cpp-ranges-chunk-view-inner-iterator:0006 */}
```cpp
#include <iostream>
#include <iterator>
#include <ranges>
#include <sstream>
 
int main()
{
    auto letters = std::istringstream{"ABCDEFGHIJK"};
 
    auto chunks = std::ranges::istream_view<char>(letters)
                | std::views::chunk(4);
 
    for (auto chunk : chunks)
    {
        // chunk is an object of type chunk_view::outer_iterator::value_type
        std::cout << '[';
        for (auto inner_iter = chunk.begin(); inner_iter != std::default_sentinel;
             ++inner_iter)
            std::cout << *inner_iter;
        std::cout << "] ";
    }
    std::cout << '\n';
}
```

Output:

```text
[ABCD] [EFGH] [IJK]
```
