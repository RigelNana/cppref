{/* source:cpp-string-basic-string-view-remove-prefix:0001 */}
Moves the start of the view forward by `n` characters.

{/* source:cpp-string-basic-string-view-remove-prefix:0002 */}
The behavior is undefined if `n > size()`.

### Parameters

{/* source:cpp-string-basic-string-view-remove-prefix:0003 */}
<ParameterList>
  <Parameter name="n">number of characters to remove from the start of the view</Parameter>
</ParameterList>

### Return value

{/* source:cpp-string-basic-string-view-remove-prefix:0004 */}
(none)

### Complexity

{/* source:cpp-string-basic-string-view-remove-prefix:0005 */}
Constant.

### Example

{/* source:cpp-string-basic-string-view-remove-prefix:0006 */}
```cpp
#include <algorithm>
#include <iostream>
#include <string_view>
 
using namespace std::literals;
 
[[nodiscard("a pure function")]]
constexpr std::size_t count_substrings(std::string_view hive, std::string_view const bee)
{
    if (hive.empty() || bee.empty())
        return 0U;
 
    std::size_t buzz{};
    while (bee.size() <= hive.size())
    {
        const auto pos = hive.find(bee);
        if (pos == hive.npos)
            break;
        ++buzz;
        hive.remove_prefix(pos + bee.size());
    }
    return buzz;
}
 
int main()
{
    std::string str = "   trim me";
    std::string_view v = str;
    v.remove_prefix(std::min(v.find_first_not_of(" "), v.size()));
    std::cout << "String: '" << str << "'\n"
              << "View  : '" << v << "'\n";
 
    constexpr auto hive{"bee buzz bee buzz bee"};
    std::cout << "There are " << count_substrings(hive, "bee") << " bees in this hive.\n";
}
```

Output:

```text
String: '   trim me'
View  : 'trim me'
There are 3 bees in this hive.
```
