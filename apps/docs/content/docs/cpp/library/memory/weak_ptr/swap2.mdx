Specializes the [`std::swap`](/docs/cpp/library/algorithm/swap) algorithm for `std::weak_ptr`. Swaps the contents of `lhs` and `rhs`. Calls `lhs.swap(rhs)`.

### Parameters

{/* source:cpp-memory-weak-ptr-swap2:0002 */}
<ParameterList>
  <Parameter name="lhs, rhs">smart pointers whose contents to swap</Parameter>
</ParameterList>

### Return value

{/* source:cpp-memory-weak-ptr-swap2:0003 */}
(none)

### Complexity

{/* source:cpp-memory-weak-ptr-swap2:0004 */}
Constant.

### Example

{/* source:cpp-memory-weak-ptr-swap2:0005 */}
```cpp
#include <iostream>
#include <memory>
#include <string>
 
struct Foo {
    Foo(int _val) : val(_val) { std::cout << "Foo...\n"; }
    ~Foo() { std::cout << "~Foo...\n"; }
    std::string print() { return std::to_string(val); }
    int val;
};
 
int main()
{
    std::shared_ptr<Foo> sp1 = std::make_shared<Foo>(100);
    std::shared_ptr<Foo> sp2 = std::make_shared<Foo>(200);
    std::weak_ptr<Foo> wp1 = sp1;
    std::weak_ptr<Foo> wp2 = sp2;
    auto print = [&]() {
        auto p1 = wp1.lock();
        auto p2 = wp2.lock();
        std::cout << " p1=" << (p1 ? p1->print() : "nullptr");
        std::cout << " p2=" << (p2 ? p2->print() : "nullptr") << '\n';  
    };
    print();
 
    std::swap(wp1, wp2);
    print();
 
    wp1.reset();
    print();
 
    std::swap(wp1, wp2);
    print();   
}
```

Output:

```text
Foo...
Foo...
 p1=100 p2=200
 p1=200 p2=100
 p1=nullptr p2=100
 p1=100 p2=nullptr
~Foo...
~Foo...
```
