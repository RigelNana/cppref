{/* source:cpp-utility-functional-invoke:0001 */}
1. 1) Invoke the *Callable* object `f` with the parameters `args` as by [*`INVOKE`*](/docs/cpp/library/utility/functional)`(std::forward<F>(f), std::forward<Args>(args)...)`. This overload participates in overload resolution only if `std::is_invocable_v<F, Args...>` is `true`.

{/* source:cpp-utility-functional-invoke:0002 */}
2. 2) Invoke the *Callable* object `f` with the parameters `args` as by [*`INVOKE<R>`*](/docs/cpp/library/utility/functional)`(std::forward<F>(f), std::forward<Args>(args)...)`. This overload participates in overload resolution only if `std::is_invocable_r_v<R, F, Args...>` is `true`.

### Parameters

{/* source:cpp-utility-functional-invoke:0003 */}
<ParameterList>
  <Parameter name="f">[*Callable*](/docs/cpp/library/named_req/Callable) object to be invoked</Parameter>
  <Parameter name="args">arguments to pass to `f`</Parameter>
</ParameterList>

### Return value

{/* source:cpp-utility-functional-invoke:0004 */}
1. 1) The value returned by `f`.

{/* source:cpp-utility-functional-invoke:0005 */}
2. 2) The value returned by `f`, implicitly converted to `R`, if `R` is not (possibly [cv-qualified](/docs/cpp/language/declarations/cv)) `void`. None otherwise.

### Exceptions

{/* source:cpp-utility-functional-invoke:0006 */}
1. 1) [`noexcept`](/docs/cpp/language/exceptions/noexcept_spec) specification: `noexcept(std::is_nothrow_invocable_v<F, Args...>)`

{/* source:cpp-utility-functional-invoke:0007 */}
2. 2) [`noexcept`](/docs/cpp/language/exceptions/noexcept_spec) specification: `noexcept(std::is_nothrow_invocable_r_v<R, F, Args...>)`

### Possible implementation

{/* source:cpp-utility-functional-invoke:0008 */}
[invoke (1)](/docs/cpp/library/utility/functional/invoke#Version_1)

```cpp
namespace detail
{
    template<class>
    constexpr bool is_reference_wrapper_v = false;
    template<class U>
    constexpr bool is_reference_wrapper_v<std::reference_wrapper<U>> = true;
 
    template<class T>
    using remove_cvref_t = std::remove_cv_t<std::remove_reference_t<T>>;
 
    template<class C, class Pointed, class Object, class... Args>
    constexpr decltype(auto) invoke_memptr(Pointed C::* member, Object&& object,
                                           Args&&... args)
    {
        using object_t = remove_cvref_t<Object>;
        constexpr bool is_member_function = std::is_function_v<Pointed>;
        constexpr bool is_wrapped = is_reference_wrapper_v<object_t>;
        constexpr bool is_derived_object = std::is_same_v<C, object_t>
                                        || std::is_base_of_v<C, object_t>;
 
        if constexpr (is_member_function)
        {
            if constexpr (is_derived_object)
                return (std::forward<Object>(object) .* member)
                           (std::forward<Args>(args)...);
            else if constexpr (is_wrapped)
                return (object.get() .* member)(std::forward<Args>(args)...);
            else
                return ((*std::forward<Object>(object)) .* member)
                           (std::forward<Args>(args)...);
        }
        else
        {
            static_assert(std::is_object_v<Pointed> && sizeof...(args) == 0);
            if constexpr (is_derived_object)
                return std::forward<Object>(object) .* member;
            else if constexpr (is_wrapped)
                return object.get() .* member;
            else
                return (*std::forward<Object>(object)) .* member;
        }
    }
} // namespace detail
 
template<class F, class... Args>
constexpr std::invoke_result_t<F, Args...> invoke(F&& f, Args&&... args)
    noexcept(std::is_nothrow_invocable_v<F, Args...>)
{
    if constexpr (std::is_member_pointer_v<detail::remove_cvref_t<F>>)
        return detail::invoke_memptr(f, std::forward<Args>(args)...);
    else
        return std::forward<F>(f)(std::forward<Args>(args)...);
}
```

[invoke_r (2)](/docs/cpp/library/utility/functional/invoke#Version_2)

```cpp
template<class R, class F, class... Args>
    requires std::is_invocable_r_v<R, F, Args...>
constexpr R invoke_r(F&& f, Args&&... args)
    noexcept(std::is_nothrow_invocable_r_v<R, F, Args...>)
{
    if constexpr (std::is_void_v<R>)
        std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
    else
        return std::invoke(std::forward<F>(f), std::forward<Args>(args)...);
}
```

### Example

{/* source:cpp-utility-functional-invoke:0010 */}
```cpp
#include <functional>
#include <iostream>
#include <type_traits>
 
struct Foo
{
    Foo(int num) : num_(num) {}
    void print_add(int i) const { std::cout << num_ + i << '\n'; }
    int num_;
};
 
void print_num(int i)
{
    std::cout << i << '\n';
}
 
struct PrintNum
{
    void operator()(int i) const
    {
        std::cout << i << '\n';
    }
};
 
int main()
{
    std::cout << "invoke a free function: ";
    std::invoke(print_num, -9);
 
    std::cout << "invoke a lambda: ";
    std::invoke([](){ print_num(42); });
 
    std::cout << "invoke a member function: ";
    const Foo foo(314159);
    std::invoke(&Foo::print_add, foo, 1);
 
    std::cout << "invoke (i.e., access) a data member num_: "
              << std::invoke(&Foo::num_, foo) << '\n';
 
    std::cout << "invoke a function object: ";
    std::invoke(PrintNum(), 18);
 
#if defined(__cpp_lib_invoke_r)
    auto add = [](int x, int y){ return x + y; };
    std::cout << "invoke a lambda converting result to float: ";
    auto ret = std::invoke_r<float>(add, 11, 22);
    static_assert(std::is_same<decltype(ret), float>());
    std::cout << std::fixed << ret << "\ninvoke print_num: ";
    std::invoke_r<void>(print_num, 44);
#endif
}
```

Possible output:

```text
invoke a free function: -9
invoke a lambda: 42
invoke a member function: 314160
invoke (i.e., access) a data member num_: 314159
invoke a function object: 18
invoke a lambda converting result to float: 33.000000
invoke print_num: 44
```
