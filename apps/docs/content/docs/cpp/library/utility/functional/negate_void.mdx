### Example

{/* source:cpp-utility-functional-negate-void:0005 */}
```cpp
#include <complex>
#include <functional>
#include <iostream>
 
int main()
{
    auto complex_negate = std::negate<void>{}; // “void” can be omitted
    constexpr std::complex z(4, 2);
    std::cout << z << '\n';
    std::cout << -z << '\n';
    std::cout << complex_negate(z) << '\n';
}
```

Output:

```text
(4,2)
(-4,-2)
(-4,-2)
```
