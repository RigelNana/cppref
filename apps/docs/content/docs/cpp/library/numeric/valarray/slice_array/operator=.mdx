{/* source:cpp-numeric-valarray-slice-array-operator:0001 */}
Assigns values to all referred elements.

{/* source:cpp-numeric-valarray-slice-array-operator:0002 */}
1) Assigns `value` to all of the elements.

{/* source:cpp-numeric-valarray-slice-array-operator:0003 */}
2) Assigns the elements of `val_arr` to the referred to elements of `*this`.

{/* source:cpp-numeric-valarray-slice-array-operator:0004 */}
3) Assigns the selected elements from `other_arr` to the referred to elements of `*this`.

### Parameters

{/* source:cpp-numeric-valarray-slice-array-operator:0005 */}
<ParameterList>
  <Parameter name="value">a value to assign to all of the referred elements</Parameter>
  <Parameter name="val_arr">[`std::valarray`](/docs/cpp/library/numeric/valarray) to assign</Parameter>
  <Parameter name="other_arr">[`std::slice_array`](/docs/cpp/library/numeric/valarray/slice_array) to assign</Parameter>
</ParameterList>

### Return value

{/* source:cpp-numeric-valarray-slice-array-operator:0006 */}
1,2) (none)

{/* source:cpp-numeric-valarray-slice-array-operator:0007 */}
3) `*this`

### Example

{/* source:cpp-numeric-valarray-slice-array-operator:0008 */}
Output:
```cpp
#include <iostream>
#include <valarray>
 
void print(char const* remark, std::valarray<int> const& v = {})
{
    std::cout << remark;
    if (v.size() != 0)
    {
        std::cout << ':';
        for (int e : v)
            std::cout << ' ' << e;
    }
    std::cout << '\n';
}
 
int main()
{
    std::valarray<int> v1{1, 2, 3, 4, 5, 6, 7, 8};
    std::slice_array<int> s1 = v1[std::slice(1, 4, 2)];
    print("v1", v1);
    print("s1", s1);
 
    print("\n(1) operator=( const T& )");
    print("s1 = 42");
    s1 = 42; // (1)
    print("s1", s1);
    print("v1", v1);
 
    print("\n(2) operator=( const std::valarray<T>& )");
    std::valarray<int> v2{10, 11, 12, 13, 14, 15};
    print("v2", v2);
    print("s1 = v2");
    s1 = v2; // (2)
    print("s1", s1);
    print("v1", v1);
 
    print("\n(3) operator=( const slice_array& )");
    v1 = {1, 2, 3, 4, 5, 6, 7, 8};
    std::slice_array<int> s2 = v1[std::slice(0, 4, 1)];
    std::slice_array<int> s3 = v2[std::slice(1, 4, 1)];
    print("v1", v1);
    print("v2", v2);
    print("s2", s2);
    print("s3", s3);
    print("s2 = s3");
    s2 = s3; // (3) Note: LHS and RHS must have same size.
    print("s2", s2);
    print("v1", v1);
}
```
