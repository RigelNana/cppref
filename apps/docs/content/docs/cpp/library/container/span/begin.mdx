{/* source:cpp-container-span-begin:0001 */}
Returns an iterator to the first element of the `span`.

{/* source:cpp-container-span-begin:0002 */}
If the `span` is empty, the returned iterator will be equal to `end()`.

## Parameters

{/* source:cpp-container-span-begin:0003 */}
(none)

## Return value

{/* source:cpp-container-span-begin:0004 */}
Iterator to the first element.

## Complexity

{/* source:cpp-container-span-begin:0005 */}
Constant.

## Example

{/* source:cpp-container-span-begin:0006 */}
```cpp
#include <iostream>
#include <span>

void print(std::span<const int> array)
{
    std::cout << "array = ";
    for (auto it = array.begin(); it != array.end(); ++it)
        std::cout << *it << ' ';
    std::cout << '\n';
}

void set_first_element(std::span<int> sp, int new_value)
{
    if (!sp.empty())
    {
        std::cout << "old *begin = " << *sp.begin() << '\n';
        *sp.begin() = new_value;
        std::cout << "new *begin = " << *sp.begin() << '\n';
    }
}

int main()
{
    int array[]{1, 3, 4, 5};
    print(array);
    set_first_element(array, 2);
    print(array);
}
```

Output:

```text
array = 1 3 4 5
old *begin = 1
new *begin = 2
array = 2 3 4 5
```
